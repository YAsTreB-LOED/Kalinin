@echo off
chcp 65001 >nul
setlocal

set "PS1=%TEMP%\smb_check_user_%RANDOM%.ps1"

> "%PS1%" echo function Get-RegValue {
>> "%PS1%" echo     param($Path, $Name)
>> "%PS1%" echo     try {
>> "%PS1%" echo         $v = Get-ItemProperty -Path $Path -Name $Name -ErrorAction Stop
>> "%PS1%" echo         return $v.$Name
>> "%PS1%" echo     } catch { return $null }
>> "%PS1%" echo }
>> "%PS1%" echo.
>> "%PS1%" echo $srvParams = "HKLM:\SYSTEM\CurrentControlSet\Services\LanmanServer\Parameters"
>> "%PS1%" echo $wksParams = "HKLM:\SYSTEM\CurrentControlSet\Services\LanmanWorkstation\Parameters"
>> "%PS1%" echo.
>> "%PS1%" echo $smb1Srv = Get-RegValue $srvParams "SMB1"
>> "%PS1%" echo $smb1Wks = Get-RegValue $wksParams "SMB1"
>> "%PS1%" echo $signSrv = Get-RegValue $srvParams "RequireSecuritySignature"
>> "%PS1%" echo $signWks = Get-RegValue $wksParams "RequireSecuritySignature"
>> "%PS1%" echo.
>> "%PS1%" echo Write-Host "=== SMBv1 ===" -ForegroundColor Cyan
>> "%PS1%" echo if ($smb1Srv -eq $null) { Write-Host "Сервер: не задано (по умолчанию отключен в Windows 10 1709+)" } else { Write-Host "Сервер: SMB1 = $smb1Srv" }
>> "%PS1%" echo if ($smb1Wks -eq $null) { Write-Host "Клиент: не задано (по умолчанию отключен в Windows 10 1709+)" } else { Write-Host "Клиент: SMB1 = $smb1Wks" }
>> "%PS1%" echo if ($smb1Srv -eq 1 -or $smb1Wks -eq 1) { Write-Host "РЕКОМЕНДАЦИЯ: SMBv1 включен явно - отключите!" -ForegroundColor Red } else { Write-Host "OK: SMBv1 не включен принудительно." -ForegroundColor Green }
>> "%PS1%" echo.
>> "%PS1%" echo Write-Host "=== Подпись SMB ===" -ForegroundColor Cyan
>> "%PS1%" echo if ($signSrv -eq $null) { Write-Host "Сервер: RequireSecuritySignature не задано (по умолчанию 0)" } else { Write-Host "Сервер: RequireSecuritySignature = $signSrv" }
>> "%PS1%" echo if ($signWks -eq $null) { Write-Host "Клиент: RequireSecuritySignature не задано (по умолчанию 0)" } else { Write-Host "Клиент: RequireSecuritySignature = $signWks" }
>> "%PS1%" echo if ($signSrv -ne 1 -or $signWks -ne 1) { Write-Host "РЕКОМЕНДАЦИЯ: включите обязательную подпись SMB (требует админа)." -ForegroundColor Yellow } else { Write-Host "OK: подпись SMB требуется." -ForegroundColor Green }
>> "%PS1%" echo.
>> "%PS1%" echo Write-Host "=== Активные SMB-соединения ===" -ForegroundColor Cyan
>> "%PS1%" echo $conn = Get-SmbConnection -ErrorAction SilentlyContinue
>> "%PS1%" echo if ($conn) { $conn ^| Select-Object ServerName, ShareName, Dialect ^| Format-Table -AutoSize ^| Out-String ^| Write-Host } else { Write-Host "Нет активных SMB-подключений или модуль SmbShare недоступен." }
>> "%PS1%" echo.
>> "%PS1%" echo Write-Host "Примечание: полная проверка (Get-SmbServerConfiguration, DISM) требует прав администратора." -ForegroundColor DarkGray

powershell -NoProfile -ExecutionPolicy Bypass -File "%PS1%"
del "%PS1%"

echo.
pause
